# 2026-10-19 OOM on dev-u24: a tree configured with NO build type compiles with
# neither -O nor -g, and the objects balloon with mangled template symbol names
# (85 MB per TU; libwilleq_core.a 5.7 GB against 130 MB for Release). A full
# `cmake --build -j12` of that tree then ran ten links at once, 38 GB on a
# 32 GB box, and the kernel OOM killer took two sessions with it. An empty
# build type therefore means Release; pass -DCMAKE_BUILD_TYPE explicitly for
# anything else.
IF(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    SET(CMAKE_BUILD_TYPE Release CACHE STRING
        "Build type (empty defaulted to Release, see the 2026-10-19 note)" FORCE)
ENDIF()

# Find Irrlicht (required for graphics rendering, when EQT_GRAPHICS=ON).
#
# P30 (Phase 3, desktop_gl_elimination_plan.md): the system-Irrlicht
# FIND_PATH/FIND_LIBRARY fallback (searching for an apt-installed
# libirrlicht-dev, i.e. desktop GL / X11-GLX) was REMOVED — that backend no
# longer exists in this project. EQT_GLES2_IRRLICHT_DIR (a locally-built
# patched GLES2 Irrlicht, see scripts/build-gles2-x86.sh) is now the only
# supported way to provide Irrlicht for a from-scratch configure.
#
# ARM cross-compiles (docker/Dockerfile.arm-noble, scripts/build-arm-noble.sh)
# don't set EQT_GLES2_IRRLICHT_DIR at all — they pass -DIRRLICHT_INCLUDE_DIR /
# -DIRRLICHT_LIBRARY directly as CMake cache variables pointing at the
# ARM sysroot's own patched Irrlicht (installed via `cmake --install` earlier
# in the Dockerfile). Both mechanisms funnel into the same IRRLICHT_INCLUDE_DIR
# / IRRLICHT_LIBRARY variables checked below, so the ARM path is unaffected by
# this removal.
IF(EQT_GRAPHICS)
    IF(EQT_GLES2_IRRLICHT_DIR)
        # H01: use the locally-built patched Irrlicht (x86 GLES2 dev builds)
        # instead of searching the system for one.
        # Absolutize first: CMake's EXISTS is only defined for absolute
        # paths, so the documented relative form
        # (-DEQT_GLES2_IRRLICHT_DIR=third_party/irrlicht-gles2-x86) would
        # otherwise fail the check below even when the library is present.
        GET_FILENAME_COMPONENT(EQT_GLES2_IRRLICHT_DIR_ABS
            "${EQT_GLES2_IRRLICHT_DIR}" ABSOLUTE
            BASE_DIR "${CMAKE_SOURCE_DIR}")
        # Second trap: CMake itself absolutizes an untyped -D value against
        # the *invocation* cwd when the cache entry is typed PATH — so a
        # relative value passed while running `cmake -S` from elsewhere
        # arrives here already absolute-but-wrong. If the resolved location
        # lacks the library but the in-tree default has it, prefer the
        # in-tree default (with a notice).
        IF(NOT EXISTS "${EQT_GLES2_IRRLICHT_DIR_ABS}/lib/libIrrlicht.a"
           AND EXISTS "${CMAKE_SOURCE_DIR}/third_party/irrlicht-gles2-x86/lib/libIrrlicht.a")
            MESSAGE(WARNING "EQT_GLES2_IRRLICHT_DIR resolved to "
                "'${EQT_GLES2_IRRLICHT_DIR_ABS}' (no libIrrlicht.a there); "
                "using the in-tree default third_party/irrlicht-gles2-x86 "
                "instead. If you intended an alternate Irrlicht tree, pass "
                "an absolute path to it.")
            SET(EQT_GLES2_IRRLICHT_DIR_ABS
                "${CMAKE_SOURCE_DIR}/third_party/irrlicht-gles2-x86")
        ENDIF()
        SET(IRRLICHT_INCLUDE_DIR "${EQT_GLES2_IRRLICHT_DIR_ABS}/include")
        SET(IRRLICHT_LIBRARY "${EQT_GLES2_IRRLICHT_DIR_ABS}/lib/libIrrlicht.a")
    ENDIF()
    IF(IRRLICHT_INCLUDE_DIR AND IRRLICHT_LIBRARY AND EXISTS "${IRRLICHT_INCLUDE_DIR}/irrlicht.h" AND EXISTS "${IRRLICHT_LIBRARY}")
        ADD_DEFINITIONS(-DEQT_HAS_GRAPHICS)
        MESSAGE(STATUS "Irrlicht graphics support: ENABLED (${IRRLICHT_LIBRARY})")
        IF(NOT (EQT_DRM OR EQT_GLES2))
            MESSAGE(FATAL_ERROR
                "EQT_GRAPHICS=ON with a patched Irrlicht requires -DEQT_GLES2=ON "
                "(x86 GLES2 dev builds) or -DEQT_DRM=ON (ARM DRM/KMS builds) — "
                "desktop GL (X11/GLX) support was removed (P30, "
                "refactor/desktop_gl_elimination_plan.md). Configure with "
                "-DEQT_GLES2=ON.")
        ENDIF()
    ELSE()
        MESSAGE(FATAL_ERROR
            "EQT_GRAPHICS=ON requires a locally-built patched GLES2 Irrlicht — "
            "desktop-GL Irrlicht (system libirrlicht-dev / X11-GLX) support was "
            "removed (P30, refactor/desktop_gl_elimination_plan.md). "
            "For x86 dev builds: run 'bash scripts/build-gles2-x86.sh "
            "--irrlicht-only' once, then configure with "
            "-DEQT_GLES2_IRRLICHT_DIR=third_party/irrlicht-gles2-x86 "
            "-DEQT_GLES2=ON (see CLAUDE.md 'Build Commands'). For ARM "
            "cross-compiles, use scripts/build-arm-noble.sh, which supplies "
            "IRRLICHT_INCLUDE_DIR/IRRLICHT_LIBRARY directly. To build without "
            "graphics at all, configure with -DEQT_GRAPHICS=OFF.")
    ENDIF()
ENDIF()

# B1043 (#363): macOS GLES2 = ANGLE (EGL + GLES2 over Metal). macOS has no
# system EGL/GLESv2, libdrm, GBM or X11, and every GLES2 target in this file
# links the literal names `EGL GLESv2 drm gbm` (+ `X11` on x86). On Darwin
# those names become CMake targets: EGL/GLESv2 = ANGLE's dylibs, drm/gbm/X11 =
# empty INTERFACE libraries (the patched Irrlicht's Darwin device has no
# DRM/GBM/X11 code -- docker/irrlicht-drm-x86/CIrrDeviceFB_Darwin.cpp; its
# window is Cocoa, CIrrDeviceFB_Cocoa.mm). No
# Linux/ARM configure sees any of this. Recipe: docs/macos.md,
# scripts/build-gles2-darwin.sh.
IF(APPLE AND EQT_GRAPHICS AND EQT_GLES2)
    SET(EQT_ANGLE_DIR "" CACHE PATH "ANGLE source checkout (headers in <dir>/include) for macOS GLES2 builds")
    SET(EQT_ANGLE_LIB_DIR "" CACHE PATH "Directory holding ANGLE's libEGL.dylib + libGLESv2.dylib")
    IF(NOT EXISTS "${EQT_ANGLE_DIR}/include/EGL/egl.h"
       OR NOT EXISTS "${EQT_ANGLE_LIB_DIR}/libEGL.dylib"
       OR NOT EXISTS "${EQT_ANGLE_LIB_DIR}/libGLESv2.dylib")
        MESSAGE(FATAL_ERROR "macOS EQT_GLES2=ON needs ANGLE: -DEQT_ANGLE_DIR=<angle checkout> "
            "-DEQT_ANGLE_LIB_DIR=<angle>/out/Release (docs/macos.md)")
    ENDIF()
    INCLUDE_DIRECTORIES(SYSTEM "${EQT_ANGLE_DIR}/include")
    ADD_LIBRARY(EGL SHARED IMPORTED)
    # B1044: every target that links the patched Irrlicht links `EGL`, and the
    # archive's device now carries the Cocoa window backend
    # (CIrrDeviceFB_Cocoa.mm: AppKit, CAMetalLayer, CGSession), so the
    # frameworks ride on EGL rather than on every tool's own link line.
    SET_TARGET_PROPERTIES(EGL PROPERTIES IMPORTED_LOCATION "${EQT_ANGLE_LIB_DIR}/libEGL.dylib"
        INTERFACE_LINK_LIBRARIES "-framework Cocoa;-framework QuartzCore;-framework CoreGraphics")
    ADD_LIBRARY(GLESv2 SHARED IMPORTED)
    SET_TARGET_PROPERTIES(GLESv2 PROPERTIES IMPORTED_LOCATION "${EQT_ANGLE_LIB_DIR}/libGLESv2.dylib")
    ADD_LIBRARY(drm INTERFACE)
    ADD_LIBRARY(gbm INTERFACE)
    ADD_LIBRARY(X11 INTERFACE)
    # ANGLE's dylibs are @rpath/-named; run from the build tree without
    # DYLD_LIBRARY_PATH.
    LIST(APPEND CMAKE_BUILD_RPATH "${EQT_ANGLE_LIB_DIR}")
    MESSAGE(STATUS "macOS GLES2: ANGLE ${EQT_ANGLE_LIB_DIR}")
ENDIF()

# B950 (GitLab #268): PUBLIC, not PRIVATE. ZoneMeshBuilder's member layout in
# zone_geometry.h USED to depend on EQT_HAS_TEXTURE_ATLAS (the two #ifdef
# data-member blocks -- the shaderMaterialAtlas* members -- changed sizeof,
# 8 bytes total on x86-64), so
# every TU that constructs one must see the same
# layout the core's constructor was generated against. PRIVATE left every
# consumer of willeq_core one sizeof smaller (B944's ASan stack-buffer-
# overflow; test_placed_object_meshes by construction). B494 (this MR)
# deleted the guarded members: zone_geometry.h now has zero #ifdef blocks,
# so the define gates nothing in the header and the layout no longer depends
# on it. The define and the net are KEPT but vacuous (retirement is a
# FOLLOW-UP); the net still asserts consumer and core agree. No target links
# willeq_core and compiles zone_geometry.cpp itself, so the define
# propagate through the link cannot shadow a local copy -- the per-target
# PRIVATE copies in the three tests that linked willeq_core
# (renderer_guards.cmake) are deleted as now-redundant. The three gles2
# tools below still declare it PRIVATE: they do NOT link willeq_core and
# compile the zone TUs themselves (zone_geometry.cpp, mesh_building.cpp),
# so for them the PRIVATE copy is the only
# one and stays.
# Pinned by test_zone_geometry_atlas_layout.
TARGET_COMPILE_DEFINITIONS(willeq_core PUBLIC EQT_HAS_TEXTURE_ATLAS)
# B1079 (#399): eqe1_loader.cpp compiles inside willeq_core WITHOUT
# EQT_HAS_LOGGING, so its lines were raw fprintf -- stderr ERROR immediately,
# stdout INFO only when stdout flushed (the 2026-10-15 "last line is an
# ERROR" misread was that buffering, #399 note 3). With the define, every
# message goes through the logger: one ordered stream, and the --log-file
# sink sees them. The standalone test tools (test_eqe1_loader,
# test_eqe1_stale_pack) do not link willeq_core and compile eqe1_loader.cpp
# themselves -- they keep the fprintf path by definition.
TARGET_COMPILE_DEFINITIONS(willeq_core PRIVATE EQT_HAS_LOGGING)
# SQLite for WilleqDatabase
TARGET_INCLUDE_DIRECTORIES(willeq_core PRIVATE ${CMAKE_SOURCE_DIR}/third_party)
TARGET_COMPILE_DEFINITIONS(willeq_core PRIVATE SQLITE_THREADSAFE=1 SQLITE_OMIT_LOAD_EXTENSION)
